window.saveMatchScore = function() { let matchId = document.getElementById('modal-match-id').value; let m = state.matches.find(x => x.id === matchId); if (!m) return; let rawP1 = document.getElementById('set-p1').value.trim(); let rawP2 = document.getElementById('set-p2').value.trim(); let rawTBP1 = document.getElementById('tb-p1').value.trim(); let rawTBP2 = document.getElementById('tb-p2').value.trim(); if (rawP1 === '' || rawP2 === '' || isNaN(Number(rawP1)) || isNaN(Number(rawP2))) { showToast("Please enter valid game scores for both players.", "error"); return; } let scoreP1 = Number(rawP1); let scoreP2 = Number(rawP2); let tbP1 = rawTBP1 === '' ? null : Number(rawTBP1); let tbP2 = rawTBP2 === '' ? null : Number(rawTBP2); if ((scoreP1 === 6 && scoreP2 === 6) || (scoreP1 === 7 && scoreP2 === 6) || (scoreP1 === 6 && scoreP2 === 7)) { if (tbP1 === null || tbP2 === null) { showToast("A tie-break score is required when the set reaches 6-6.", "error"); return; } const winner = scoreP1 > scoreP2 ? { games: scoreP1, tb: tbP1 } : { games: scoreP2, tb: tbP2 }; const loser = scoreP1 > scoreP2 ? { games: scoreP2, tb: tbP2 } : { games: scoreP1, tb: tbP1 }; if (winner.tb < 7 || Math.abs(winner.tb - loser.tb) < 2) { showToast("Tie-break winner must reach at least 7 points and lead by 2.", "error"); return; } if (scoreP1 > scoreP2) { scoreP1 = 7; scoreP2 = 6; } else { scoreP1 = 6; scoreP2 = 7; } } else if (scoreP1 === scoreP2) { showToast("Tennis matches cannot end in a tie. One player must win more games.", "error"); return; } else if (Math.min(scoreP1, scoreP2) < 0 || Math.max(scoreP1, scoreP2) > 15) { showToast("Set scores must be between 0 and 15.", "error"); return; } if (Math.max(scoreP1, scoreP2) >= 7 && Math.abs(scoreP1 - scoreP2) < 2) { showToast("If a set goes to 7+, the winner must lead by at least 2 games (for example 7-5 or 7-6).", "error"); return; } m.scoreP1 = scoreP1; m.scoreP2 = scoreP2; m.tiebreakP1 = tbP1; m.tiebreakP2 = tbP2; m.completed = true; persistState(); closeScoreModal(); showToast("Single-set match result saved & synced!", "success"); };
