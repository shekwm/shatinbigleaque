window.saveMatchScore = function() { let matchId = document.getElementById('modal-match-id').value; let m = state.matches.find(x => x.id === matchId); if (!m) return; let rawP1 = document.getElementById('set-p1').value.trim(); let rawP2 = document.getElementById('set-p2').value.trim(); let rawTBP1 = document.getElementById('tb-p1').value.trim(); let rawTBP2 = document.getElementById('tb-p2').value.trim(); if (rawP1 === '' || rawP2 === '' || isNaN(Number(rawP1)) || isNaN(Number(rawP2))) { showToast("Please enter valid game scores for both players.", "error"); return; } let scoreP1 = Number(rawP1); let scoreP2 = Number(rawP2); let tbP1 = rawTBP1 === '' ? null : Number(rawTBP1); let tbP2 = rawTBP2 === '' ? null : Number(rawTBP2); if (!Number.isFinite(scoreP1) || !Number.isFinite(scoreP2)) { showToast("Scores must be valid numbers.", "error"); return; } if ((tbP1 !== null && !Number.isFinite(tbP1)) || (tbP2 !== null && !Number.isFinite(tbP2))) { showToast("Tie-break values must be valid numbers when provided.", "error"); return; } m.scoreP1 = scoreP1; m.scoreP2 = scoreP2; m.tiebreakP1 = tbP1; m.tiebreakP2 = tbP2; m.completed = true; persistState(); closeScoreModal(); showToast("Single-set match result saved & synced!", "success"); };
